// NY TIME & SESSION
// ═══════════════════════════════════════════════════════════════════════
f_ny_hm() =>
    int h = 0
    int m = 0
    if tz_mode == "Auto (DST)"
        h := hour(timenow, "America/New_York")
        m := minute(timenow, "America/New_York")
    else if tz_mode == "Manual UTC-4 (EDT)"
        h := (hour(timenow, "UTC") - 4 + 24) % 24
        m := minute(timenow, "UTC")
    else
        h := (hour(timenow, "UTC") - 5 + 24) % 24
        m := minute(timenow, "UTC")
    [h, m]

f_session() =>
    [h, m] = f_ny_hm()
    int mins = h * 60 + m
    string kz = "Out of KZ"
    string st = "Out-KZ"
    if mins >= 1200 and mins < 1440
        kz := "Asia KZ"
        st := (1440 - mins) <= 15 ? "Warning" : "In-KZ"
    else if mins >= 120 and mins < 300
        kz := "London KZ"
        st := (300 - mins) <= 15 ? "Warning" : "In-KZ"
    else if mins >= 510 and mins < 660
        kz := "NY AM"
        st := (660 - mins) <= 15 ? "Warning" : "In-KZ"
    else if mins >= 810 and mins < 960
        kz := "NY PM"
        st := (960 - mins) <= 15 ? "Warning" : "In-KZ"
    [kz, st]

f_fmt_dur(n) =>
    int safe_n = math.max(n, 0)
    int h2 = int(safe_n / 60)
    int m2 = safe_n - h2 * 60
    h2 > 0 ? str.tostring(h2) + "h " + str.tostring(m2) + "m" : str.tostring(m2) + "m"

// Bar-time killzone check — v3.0: uses BAR time (not timenow) so the
// session gate stays consistent on historical bars & strategy backtest.
// Killzones (NY time): London 02:00–05:00 · NY AM 08:30–11:00 · NY PM 13:30–16:00
f_bar_kz() =>
    int h = 0
    int m = 0
    if tz_mode == "Auto (DST)"
        h := hour(time, "America/New_York")
        m := minute(time, "America/New_York")
    else if tz_mode == "Manual UTC-4 (EDT)"
        h := (hour(time, "UTC") - 4 + 24) % 24
        m := minute(time, "UTC")
    else
        h := (hour(time, "UTC") - 5 + 24) % 24
        m := minute(time, "UTC")
    int mins = h * 60 + m
    bool in_kz = (mins >= 120 and mins < 300) or (mins >= 510 and mins < 660) or (mins >= 810 and mins < 960)
    in_kz

f_session_ctdn() =>
    [h, m] = f_ny_hm()
    int mins = h * 60 + m
    string res = ""
    if mins >= 1200 and mins < 1440
        res := "Asia KZ · " + f_fmt_dur(1440 - mins) + " left"
    else if mins >= 120 and mins < 300
        res := "London KZ · " + f_fmt_dur(300 - mins) + " left"
    else if mins >= 510 and mins < 660
        res := "NY AM · " + f_fmt_dur(660 - mins) + " left"
    else if mins >= 810 and mins < 960
        res := "NY PM · " + f_fmt_dur(960 - mins) + " left"
    else
        int tgt = 0
        string nxt = ""
        if mins < 120
            tgt := 120
            nxt := "London"
        else if mins < 510
            tgt := 510
            nxt := "NY AM"
        else if mins < 810
            tgt := 810
            nxt := "NY PM"
        else if mins < 1200
            tgt := 1200
            nxt := "Asia"
        else
            tgt := 120 + 1440
            nxt := "London"
        int diff = tgt - mins
        if diff < 0
            diff := diff + 1440
        res := nxt + " in " + f_fmt_dur(diff)
    res

f_has_above(level) =>
    bool found = false
    int sz = array.size(ief_ty)
    if not na(level) and sz > 0
        for k = 0 to sz - 1
            if array.get(ief_top, k) > level
                found := true
                break
    found

f_has_below(level) =>
    bool found = false
    int sz = array.size(ief_ty)
    if not na(level) and sz > 0
        for k = 0 to sz - 1
            if array.get(ief_bot, k) < level
                found := true
                break
    found

f_lowest_between(_b1, _b2) =>

            float _l = low[bar_index - _b]
            if na(_lo) or _l < _lo
                _lo  := _l
                _lob := _b
    [_lo, _lob]

f_highest_between(_b1, _b2) =>

    int   _hib = na
    if not na(_b1) and not na(_b2) and _b2 > _b1 + 1 and bar_index - _b1 < 490

            float _h = high[bar_index - _b]
            if na(_hi) or _h > _hi
                _hi  := _h
                _hib := _b
    [_hi, _hib]

f_score_dots(_n) =>
    string _s = ""
    for _i = 1 to 8
        _s := _s + (_i <= _n ? "●" : "○")
    _s

f_zone_conf(_dir) =>
    bool _found = false
    int  _sz = array.size(ief_ty)
    float _band = na(htf_atr) ? 0.0 : htf_atr * 0.5
    if _sz > 0
        for _k = 0 to _sz - 1
            if array.get(ief_st, _k) == 0
                int   _ty = array.get(ief_ty, _k)
                float _zt = array.get(ief_top, _k)
                float _zb = array.get(ief_bot, _k)
                if _dir == 1 and (_ty == 1 or _ty == 3)
                    if _zb <= close and _zt >= low - _band
                        _found := true
                if _dir == -1 and (_ty == 2 or _ty == 4)
                    if _zt >= close and _zb <= high + _band
                        _found := true
    _found
